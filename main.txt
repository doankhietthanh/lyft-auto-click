var popupRegion = Region.deviceReg().bottom()

var findFastParam = FParam.timeout(50).mScore(0.85) 
var reserveParam = FParam.timeout(1500).mScore(0.85)
var searchAreaFindParam = FParam.timeout(30).mScore(0.85)
var searchAreaClickParam = FParam.timeout(100).mScore(0.85)
var searchAreaVisible = false

fun goLeft() {
    var path = [SwipePoint(Point(750, 1200), 0, 20), SwipePoint(Point(500, 1200), 80, 20)]
    swipe(path)
}

fun goRight() {
    var path = [SwipePoint(Point(500, 1200), 0, 20), SwipePoint(Point(750, 1200), 80, 20)]
    swipe(path)
}

fun checkSearchArea() {
    if (searchAreaVisible) {
        if (popupRegion.find("search_this_area", searchAreaFindParam))
            return false

        searchAreaVisible = false
        return false
    }

    if (!popupRegion.click("search_this_area", searchAreaClickParam))
        return false

    searchAreaVisible = true
    return true
}

fun checkNewAvaliableRides() {
    var result = popupRegion.find("new_available_rides", findFastParam)

    if (!result)
        return false

    popupRegion.click("new_available_rides", findFastParam)

    if (popupRegion.click("btn_reserve", reserveParam)) {
        popupRegion.click("btn_reserve_confirm", reserveParam)
        return true
    }
    
    return false
}

fun clickSearchAreaAfterSwipe() {
    var attempt = 0

    while (attempt < 3) {
        if (checkSearchArea())
            return true

        wait(30)
        attempt = attempt + 1
    }

    return false
}

fun waitForNewRides(attempts) {
    var attempt = 0

    while (attempt < attempts) {
        if (checkNewAvaliableRides())
            return true

        wait(50)
        attempt = attempt + 1
    }

    return false
}

while (true) {
    goRight()

    clickSearchAreaAfterSwipe()

    if (waitForNewRides(4))
        break

    goLeft()

    clickSearchAreaAfterSwipe()

    if (waitForNewRides(4))
        break
}
