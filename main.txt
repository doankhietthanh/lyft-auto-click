var popupRegion = Region.deviceReg().bottom()

var findFastParam = FParam.timeout(100).mScore(0.85)
var reserveParam = FParam.timeout(1500).mScore(0.85)
var searchAreaFindParam = FParam.timeout(150).mScore(0.85)
var searchAreaClickParam = FParam.timeout(500).mScore(0.85)

fun goLeft() {
    var path = [SwipePoint(Point(750, 1200), 0, 20), SwipePoint(Point(500, 1200), 120, 20)]
    swipe(path)
}

fun goRight() {
    var path = [SwipePoint(Point(500, 1200), 0, 20), SwipePoint(Point(750, 1200), 120, 20)]
    swipe(path)
}

fun checkSearchArea() {
    var result = popupRegion.find("search_this_area", searchAreaFindParam)

    if (!result)
        return false

    if (popupRegion.click("search_this_area", searchAreaClickParam))
        return true

    return false
}

fun checkNewAvaliableRides() {
    var result = popupRegion.find("new_available_rides", findFastParam)

    if (!result)
        return false

    popupRegion.click("new_available_rides", findFastParam)

    if (popupRegion.click("btn_reserve", reserveParam)) {
        popupRegion.click("btn_reserve_confirm", reserveParam)
        return true
    }
    
    return false
}

fun clickSearchAreaAfterSwipe() {
    var attempt = 0

    while (attempt < 6) {
        if (checkSearchArea())
            return true

        wait(50)
        attempt = attempt + 1
    }

    return false
}

fun waitForNewRides(attempts) {
    var attempt = 0

    while (attempt < attempts) {
        if (checkNewAvaliableRides())
            return true

        wait(50)
        attempt = attempt + 1
    }

    return false
}

while (true) {
    goRight()

    clickSearchAreaAfterSwipe()

    if (waitForNewRides(8))
        break

    goLeft()

    clickSearchAreaAfterSwipe()

    if (waitForNewRides(8))
        break
}
