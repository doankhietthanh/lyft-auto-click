var popupRegion = Region.deviceReg().bottom()

var findFastParam = FParam.timeout(50).mScore(0.85)
var reserveParam = FParam.timeout(1500).mScore(0.85)
var searchAreaClickParam = FParam.timeout(50).mScore(0.85)
var pollInterval = 50
var searchAreaAttempts = 8
var availableRideAttempts = 20

fun goLeft() {
    var path = [SwipePoint(Point(750, 1200), 0, 20), SwipePoint(Point(500, 1200), 80, 20)]
    swipe(path)
}

fun goRight() {
    var path = [SwipePoint(Point(500, 1200), 0, 20), SwipePoint(Point(750, 1200), 80, 20)]
    swipe(path)
}

fun clickSearchAreaAfterSwipe() {
    var attempt = 0

    while (attempt < searchAreaAttempts) {
        if (popupRegion.click("search_this_area", searchAreaClickParam))
            return true

        wait(pollInterval)
        attempt = attempt + 1
    }

    return false
}

fun reserveAvailableRide() {
    if (!popupRegion.click("new_available_rides", findFastParam))
        return false

    if (!popupRegion.click("btn_reserve", reserveParam))
        return false

    if (!popupRegion.click("btn_reserve_confirm", reserveParam))
        return false

    return true
}

fun waitForAvailableRide() {
    var attempt = 0

    while (attempt < availableRideAttempts) {
        if (popupRegion.find("new_available_rides", findFastParam))
            return reserveAvailableRide()

        wait(pollInterval)
        attempt = attempt + 1
    }

    return false
}

fun runSwipeCycle(direction) {
    if (direction == "right")
        goRight()
    else
        goLeft()

    if (!clickSearchAreaAfterSwipe())
        return false

    return waitForAvailableRide()
}

while (true) {
    if (runSwipeCycle("right")) break
    if (runSwipeCycle("left")) break
}
