var popupRegion = Region.deviceReg().bottom()
var searchAreaRegion = Region.deviceReg()

var priorityRideParam = FParam.timeout(50).mScore(0.85)
var postSearchRideParam = FParam.timeout(700).mScore(0.85)
var btnReserveParam = FParam.timeout(500).mScore(0.85)
var btnConfirmParam = FParam.timeout(500).mScore(0.85)
var searchAreaClickParam = FParam.timeout(1200).mScore(0.85)
var swipeSettleDelay = 200

fun goLeft() {
    var path = [SwipePoint(Point(750, 1000), 0, 20), SwipePoint(Point(500, 1000), 120, 20)]
    swipe(path)
}

fun goRight() {
    var path = [SwipePoint(Point(500, 1000), 0, 20), SwipePoint(Point(750, 1000), 120, 20)]
    swipe(path)
}

fun completeReservation() {
    if (!popupRegion.click("btn_reserve", btnReserveParam))
        return false

    if (!popupRegion.click("btn_reserve_confirm", btnConfirmParam))
        return false

    return true
}

fun clickSearchAreaAfterSwipe() {
    return searchAreaRegion.click("search_this_area", searchAreaClickParam)
}

fun scanAfterSwipe() {
    wait(swipeSettleDelay)

    if (popupRegion.click("new_available_rides", priorityRideParam))
        return completeReservation()

    if (!clickSearchAreaAfterSwipe())
        return false

    if (popupRegion.click("new_available_rides", postSearchRideParam))
        return completeReservation()

    return false
}

while (true) {
    goRight()
    if (scanAfterSwipe()) break

    goLeft()
    if (scanAfterSwipe()) break
}
